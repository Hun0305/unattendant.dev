{{- /* 글 폴더(page bundle) 안의 그림 하나를 캡션과 함께 넣는다. {{< fig src="name-diagram.svg" alt="대체 텍스트" caption="설명과 출처" >}} - SVG는 본문에 그대로 넣어 사이트 색(--ink 등)과 다크 모드를 따른다. 대체 텍스트는 SVG 안의 ·<desc>에 쓴다 - 그 밖의 그림(jpg, png, webp)은 alt가 필수다 - 직접 만들지 않은 그림은 caption에 출처와 라이선스를 쓴다 (docs: site/README.md "그림") - by: 누가 만든 그림인지 (ai: AI가 그림, human: 운영자가 만들거나 찍음, capture: 화면 캡처, pd: 퍼블릭 도메인, licensed: 쓰도록 허락된 남의 그림). 글 하단 "그림 출처"가 이 값으로 그림마다 한 줄씩 만든다(post-figure-credits.html). 캡션의 설명은 그대로 쓴다 */ -}} {{- $src := .Get "src" -}} {{- $by := .Get "by" | default "" -}} {{- if and $by (not (in (slice "ai" "human" "capture" "pd" "licensed") $by)) }}{{ errorf "fig: by는 ai, human, capture, pd, licensed 중 하나다: %q (%s)" $by .Position }}{{ end -}} {{- $n := add (len (.Page.Store.Get "figs" | default slice)) 1 -}} {{- .Page.Store.Add "figs" (slice (dict "n" $n "by" $by)) -}} {{- $res := .Page.Resources.GetMatch $src -}} {{- if not $res }}{{ errorf "fig: 글 폴더에 그림이 없다: %q (%s)" $src .Position }}{{ end -}} <figure class="fig" id="fig-{{ $n }}"> {{- if eq $res.MediaType.SubType "svg" }} <div class="fig-svg">{{ $res.Content | safeHTML }}</div> {{- else }} {{- $alt := .Get "alt" | default "" }} {{- if not $alt }}{{ errorf "fig: %q에 alt가 필요하다 (%s)" $src .Position }}{{ end }} <img src="{{ $res.RelPermalink }}" width="{{ $res.Width }}" height="{{ $res.Height }}" alt="{{ $alt }}" loading="lazy" decoding="async"> {{- end }} {{- with .Get "caption" }} <figcaption>{{ . | markdownify }}</figcaption> {{- end }} </figure>
